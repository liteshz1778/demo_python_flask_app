# adding helm repo
helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
helm repo update

# list out the helm repo
helm list -n monitoring

# create namespace
kubectl create ns monitoring

# create helm repo monitoring
helm install monitoring prometheus-community/kube-prometheus-stack -n monitoring -f ./custom_kube_prometheus_stack.yml

# Get all kube resources
kubectl get all -n monitoring

# Access Prometheus UI 
kubectl port-forward service/prometheus-operated -n monitoring 9090:9090

# Access Grafana UI
kubectl port-forward service/monitoring-grafana -n monitoring 8080:80

# Access Prometheus Alert-Manager
kubectl port-forward service/alertmanager-operated -n monitoring 9093:9093

# Access the promethus, grafana & alert-manager using NodePort SVC
helm upgrade monitoring prometheus-community/kube-prometheus-stack -n monitoring -f values.yaml

# To get Grafana admin passwd
kubectl --namespace monitoring get secrets monitoring-grafana -o jsonpath="{.data.admin-password}" | base64 -d ; echo 

# uninstall/remove helm repo
helm uninstall monitoring -n monitoring

# remove ns
k delete ns monitoring
