1. Install Loki + Promtail (Kubernetes)
helm repo add grafana https://grafana.github.io/helm-charts
helm repo update
helm install promtail grafana/promtail -n monitoring -f  promtail_config.yaml
helm install loki grafana/loki --set loki.useTestSchema=true -n monitoring -f loki_values.yaml


helm uninstall loki -n monitoring
kubectl delete all -l app.kubernetes.io/name=loki -n monitoring
kubectl delete pvc -n monitoring --all
helm uninstall promtail -n monitoring


helm install loki grafana/loki -n monitoring \
  --set deploymentMode=SingleBinary \
  --set loki.useTestSchema=true \
  --set loki.storage.bucketNames.chunks=dummy \
  --set loki.storage.bucketNames.ruler=dummy \
  --set loki.storage.bucketNames.admin=dummy \
  --set singleBinary.replicas=1 \
  --set singleBinary.persistence.enabled=false \
  --set read.replicas=0 \
  --set write.replicas=0 \
  --set backend.replicas=0 \
  --set gateway.enabled=false

helm upgrade loki grafana/loki -n monitoring -f ./loki_values.yaml

http://loki-loki-stack:3100

3. View Logs in Grafana (Go to Explore → Loki)
Use queries:
a. {namespace="default"}

b. {namespace="default", pod=~"flask-webapp.*"}

c. {namespace="default"} |= "request_completed"

d.{namespace="default"} |= "ERROR"

e. {namespace="default"} 
| json 
| path="/users"

f. {namespace="default"} 
| json 
| latency_ms > 100

--------------------------------------------------- 


4. Create Grafana Dashboard
Add panels like:

a. Request Count
count_over_time({namespace="default"} |= "request_completed"[1m])

b. Error Rate
count_over_time({namespace="default"} |= "ERROR"[1m])
